### Tải transcrpit từ GENCODE
mkdir transcriptome
cd transcriptome
wget https://ftp.ebi.ac.uk/pub/databases/gencode/Gencode_human/release_49/gencode.v49.transcripts.fa.gz # tải transcriptome ver 09/2025
mkdir kallisto_index 
kallisto index -i kallisto_index/hg38_gencode49 gencode.v49.transcripts.fa.gz #tạo index cho transcriptome
